cmake_minimum_required(VERSION 3.22)
project(CassetteVstHost VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
include(patch-juce.cmake)
add_subdirectory(../vendor/juce juce)
juce_add_console_app(cassette-vst PRODUCT_NAME "cassette-vst")
target_sources(cassette-vst PRIVATE main.cpp)
target_compile_definitions(cassette-vst PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_VST=0 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0)
target_link_libraries(cassette-vst PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
